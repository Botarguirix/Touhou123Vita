cmake_minimum_required(VERSION 3.16)
option(T123_VITA "Build the Vita diagnostic application" OFF)
option(T123_AUTOTEST "Package scripted guest test" OFF)
if(T123_VITA)
  if(NOT DEFINED ENV{VITASDK})
    message(FATAL_ERROR "Set VITASDK before configuring")
  endif()
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE FILEPATH "Vita toolchain")
endif()
project(Touhou123Vita LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(t123_core legacy/src/core.c legacy/src/fight.c legacy/src/animation.c legacy/src/resources.c legacy/src/pattern.c legacy/src/input.c legacy/src/stage.c legacy/src/replay.c)
target_include_directories(t123_core PUBLIC legacy/include)
target_compile_options(t123_core PRIVATE -Wall -Wextra -Werror)
if(NOT T123_VITA)
  target_link_libraries(t123_core PUBLIC m)
endif()
if(T123_VITA)
  include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/font.rgba")
    message(FATAL_ERROR "Generate the local font first: python3 tools/font.py assets")
  endif()
  add_executable(touhou123vita legacy/src/vita_main.c legacy/src/soft2d.c)
  target_link_libraries(touhou123vita t123_core SceDisplay_stub SceSysmem_stub SceCtrl_stub m)
  vita_create_self(touhou123vita.self touhou123vita UNSAFE)
  set(EXTRA_FILES)
  foreach(ASSET font)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/${ASSET}.rgba")
      list(APPEND EXTRA_FILES FILE "${CMAKE_CURRENT_SOURCE_DIR}/assets/${ASSET}.rgba" "assets/${ASSET}.rgba")
    endif()
  endforeach()
  if(T123_AUTOTEST)
    list(APPEND EXTRA_FILES FILE "${CMAKE_CURRENT_SOURCE_DIR}/legacy/tests/auto-test.txt" "auto-test.txt")
  endif()
  vita_create_vpk(Touhou123Vita.vpk THSV12301 touhou123vita.self VERSION 00.01 NAME "Touhou123Vita Prototype" ${EXTRA_FILES})
else()
  enable_testing()
  add_executable(core_tests legacy/tests/core_tests.c)
  target_link_libraries(core_tests t123_core)
  add_test(NAME core_contract COMMAND core_tests)
  add_executable(fight_tests legacy/tests/fight_tests.c)
  target_link_libraries(fight_tests t123_core)
  add_test(NAME fight_contract COMMAND fight_tests)
  add_test(NAME remilia_world_contract COMMAND fight_tests --remilia-only)
  add_executable(arena_tests legacy/tests/arena_tests.c)
  target_link_libraries(arena_tests t123_core)
  add_test(NAME arena_world_contract COMMAND arena_tests)
  add_executable(collision_world_tests legacy/tests/collision_world_tests.c)
  target_link_libraries(collision_world_tests t123_core)
  add_test(NAME collision_world_contract COMMAND collision_world_tests)
  add_executable(animation_tests legacy/tests/animation_tests.c)
  target_link_libraries(animation_tests t123_core)
  add_test(NAME animation_contract COMMAND animation_tests)
  add_executable(resource_tests legacy/tests/resource_tests.c)
  target_link_libraries(resource_tests t123_core)
  add_test(NAME resource_contract COMMAND resource_tests)
  add_executable(pattern_tests legacy/tests/pattern_tests.c)
  target_link_libraries(pattern_tests t123_core)
  add_test(NAME pattern_contract COMMAND pattern_tests)
  add_executable(input_tests legacy/tests/input_tests.c)
  target_link_libraries(input_tests t123_core)
  add_test(NAME input_contract COMMAND input_tests)
  add_executable(stage_tests legacy/tests/stage_tests.c)
  target_link_libraries(stage_tests t123_core)
  add_test(NAME stage_contract COMMAND stage_tests)
  add_executable(replay_tests legacy/tests/replay_tests.c)
  target_link_libraries(replay_tests t123_core)
  add_test(NAME replay_contract COMMAND replay_tests)
  add_executable(resource_probe legacy/tests/resource_probe.c)
  target_link_libraries(resource_probe t123_core)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/artifacts/x86-animation.csv")
    add_test(NAME animation_x86_oracle COMMAND animation_tests "${CMAKE_CURRENT_SOURCE_DIR}/artifacts/x86-animation.csv")
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/artifacts/x86-animation-101.csv")
    add_test(NAME animation_x86_101_oracle COMMAND animation_tests "${CMAKE_CURRENT_SOURCE_DIR}/artifacts/x86-animation-101.csv")
  endif()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME tools_contract COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_*.py)
  set_tests_properties(tools_contract PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()
